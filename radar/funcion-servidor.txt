// ── Radar de lluvia de la AEMET (entrega 99) ──
// El radar nacional georreferenciado de la AEMET no se deja descargar (responde siempre «límite de uso»), así que
// se usan las imágenes de los radares regionales que cubren las fincas: Murcia y Valencia. Son GIF de 480×530 con
// el radar en el centro (punto 240,240), un punto por km (proyección equidistante desde el radar) y la escala en dBZ.
// La posición de cada radar se ajustó con las fronteras dibujadas en sus imágenes (error medio < 1 km).
// Aquí se lee cada GIF, se pasa cada punto a mm/h, se pinta con la escala de Rain Alarm, se juntan los dos radares
// en una imagen con la proyección del mapa y se guarda como PNG en el almacén «radar» (últimas 2 horas).
// La app lee «indice.json» de ese almacén y anima las imágenes.
const RAD_BUCKET = "radar";
const RAD_MIN = 130; // minutos que se guardan (2 h y un margen)
const RAD_RADARES = [
  { id: "mu", lat: 38.26375, lon: -1.19275, km: 0.99815, giro: 0.00295 }, // Murcia
  { id: "va", lat: 39.1844, lon: -0.25204, km: 0.99709, giro: 0.00226 }, // Valencia
];
const RAD_CAJA = { s: 36.05, o: -4.0, n: 41.4, e: 2.6 }; // lo que cubren los dos radares (240 km alrededor)
const RAD_ANCHO = 660; // puntos de ancho de la imagen guardada (≈ 0,9 km por punto)
// colores de la leyenda de las imágenes regionales de la AEMET → dBZ (desde)
const RAD_DBZ = {
  "0,0,252": 12, "0,148,252": 18, "0,252,252": 24, "67,131,35": 30, "0,192,0": 36, "0,255,0": 42,
  "255,255,0": 48, "255,187,0": 54, "255,127,0": 60, "255,0,0": 66, "200,0,90": 72,
};
// escala de Rain Alarm: desde cuántos mm/h se usa cada color
const RAD_ESC = [
  [0, 0x89, 0xdd, 0xde], [0.3, 0x01, 0x01, 0xd6], [0.6, 0x96, 0xf5, 0x65], [1.3, 0x4e, 0x86, 0x33],

];
let radarIntento = 0; // última vez que esta copia de la función pidió el radar a la AEMET

// dBZ → mm/h (Marshall-Palmer, Z = 200·R^1,6)
const dbzMmh = (dbz) => Math.pow(Math.pow(10, dbz / 10) / 200, 1 / 1.6);
// mm/h → número de color (1…12; 0 = sin lluvia)
function mmhCol(mm) {
  if (!(mm > 0)) return 0;
  let k = 0;
  for (let i = 0; i < RAD_ESC.length; i++) if (mm >= RAD_ESC[i][0]) k = i;
  return k + 1;
}

async function desinflar(u8) {
  const cs = new CompressionStream("deflate");
  const w = cs.writable.getWriter();
  w.write(u8); w.close();
  return new Uint8Array(await new Response(cs.readable).arrayBuffer());
}

// lee la primera imagen de un GIF → { ancho, alto, px (número de color de cada punto), pal (r,g,b de cada número) }
function leerGif(u) {
  if (u.length < 13 || String.fromCharCode(u[0], u[1], u[2]) !== "GIF") throw new Error("no es GIF");
  const ancho = u[6] | (u[7] << 8), alto = u[8] | (u[9] << 8);
  let o = 13, pal = null;
  if (u[10] & 0x80) { const n = 3 * (2 << (u[10] & 7)); pal = u.subarray(o, o + n); o += n; }
  const saltarBloques = () => { while (o < u.length && u[o]) o += u[o] + 1; o++; };
  while (o < u.length) {
    const t = u[o++];
    if (t === 0x21) { o++; saltarBloques(); continue; }
    if (t !== 0x2c) break;
    const x0 = u[o] | (u[o + 1] << 8), y0 = u[o + 2] | (u[o + 3] << 8), w = u[o + 4] | (u[o + 5] << 8), h = u[o + 6] | (u[o + 7] << 8), f = u[o + 8];
    o += 9;
    if (f & 0x80) { const n = 3 * (2 << (f & 7)); pal = u.subarray(o, o + n); o += n; }
    const minimo = u[o++];
    const trozos = [];
    while (o < u.length && u[o]) { trozos.push(u.subarray(o + 1, o + 1 + u[o])); o += u[o] + 1; }
    const datos = new Uint8Array(trozos.reduce((a, b) => a + b.length, 0));
    let p = 0;
    for (const b of trozos) { datos.set(b, p); p += b.length; }
    const idx = lzwGif(datos, minimo, w * h);
    const px = new Uint8Array(ancho * alto);
    const filas = [];
    if (f & 0x40) { for (const [ini, paso] of [[0, 8], [4, 8], [2, 4], [1, 2]]) for (let y = ini; y < h; y += paso) filas.push(y); }
    else for (let y = 0; y < h; y++) filas.push(y);
    filas.forEach((y, k) => { if (y0 + y < alto) px.set(idx.subarray(k * w, k * w + Math.min(w, ancho - x0)), (y0 + y) * ancho + x0); });
    if (!pal) throw new Error("GIF sin paleta");
    return { ancho, alto, px, pal };
  }
  throw new Error("GIF sin imagen");
}
// descompresión LZW de GIF (códigos de longitud variable, el bit menos significativo primero)
function lzwGif(src, minimo, tam) {
  const out = new Uint8Array(tam);
  const limpiar = 1 << minimo, fin = limpiar + 1;
  const pre = new Int32Array(4096), suf = new Uint8Array(4096), largo = new Uint16Array(4096);
  let ancho = minimo + 1, siguiente = fin + 1, prev = -1, o = 0, bits = 0, acum = 0, p = 0;
  for (let i = 0; i < limpiar; i++) { pre[i] = -1; suf[i] = i; largo[i] = 1; }
  const pila = new Uint8Array(4097);
  while (o < tam) {
    while (bits < ancho && p < src.length) { acum |= src[p++] << bits; bits += 8; }
    if (bits < ancho) break;
    const c = acum & ((1 << ancho) - 1);
    acum >>>= ancho; bits -= ancho;
    if (c === limpiar) { ancho = minimo + 1; siguiente = fin + 1; prev = -1; continue; }
    if (c === fin) break;
    let cod = c, primero;
    if (prev === -1) { out[o++] = suf[c]; prev = c; continue; }
    let k = 0;
    if (c >= siguiente) { cod = prev; }
    while (cod >= 0 && k < 4096) { pila[k++] = suf[cod]; cod = pre[cod]; }
    primero = pila[k - 1];
    if (c >= siguiente) { for (let i = k; i > 0; i--) pila[i] = pila[i - 1]; pila[0] = primero; k++; }
    for (let i = k - 1; i >= 0 && o < tam; i--) out[o++] = pila[i];
    if (siguiente < 4096) { pre[siguiente] = prev; suf[siguiente] = primero; largo[siguiente] = largo[prev] + 1; siguiente++; if (siguiente === 1 << ancho && ancho < 12) ancho++; }
    prev = c;
  }
  return out;
}

// hora de la imagen, leída del pie (texto «22 OCT 26295 202000»: año+día del año y hhmmss, en UTC)
// Letra de 7×8 puntos cada 9 puntos; las cifras 4, 7 y 8 no han salido aún en ninguna imagen y van dibujadas a mano.
const RAD_CIFRAS = {
  0: "01111101000001100000110010011001001100000110000010111110", 1: "00010000011000000100000010000001000000100000010000111110",
  2: "01111101000001000000100000100011100010000010000001111111", 5: "11111111000000100000011111100000001000000110000010111110",
  3: "11111110000010000010000011000000010000000110000010111110",
  6: "00111100100000100000010111101100001100000110000010111110", 9: "01111101000001100000110000110111101000000100000100111100",
};
const RAD_CIFRAS_MANO = {
  4: "00001100001010001001001000101111111000001000000100000010",
  7: "11111110000001000001000001000001000000100000010000001000", 8: "01111101000001100000101111101000001100000110000010111110",
};
function glifo(px, ancho, x) {
  let s = "";
  for (let y = 519; y < 527; y++) for (let k = 0; k < 7; k++) s += px[y * ancho + x + k] === 5 ? "1" : "0";
  return s;
}
function leerHora(g) {
  if (g.alto < 527) return 0;
  const tabla = [];
  for (const [d, t] of Object.entries(RAD_CIFRAS)) tabla.push([+d, t, true]);
  for (const [d, t] of Object.entries(RAD_CIFRAS_MANO)) tabla.push([+d, t, false]);
  const cifra = (k) => {
    const s = glifo(g.px, g.ancho, 27 + 9 * k);
    let mejor = -1, dmin = 99, exacta = false;
    for (const [d, t, ex] of tabla) { let dd = 0; for (let i = 0; i < 56; i++) if (s[i] !== t[i]) dd++; if (dd < dmin) { dmin = dd; mejor = d; exacta = ex; } }
    return dmin === 0 || (!exacta && dmin <= 8) ? mejor : -1;
  };
  const c = [25, 26, 27, 28, 29, 31, 32, 33, 34].map(cifra);
  if (c.some((x) => x < 0)) return 0;
  const anio = 2000 + c[0] * 10 + c[1], dia = c[2] * 100 + c[3] * 10 + c[4], hh = c[5] * 10 + c[6], mm = c[7] * 10 + c[8];
  if (dia < 1 || dia > 366 || hh > 23 || mm > 59) return 0;
  return Date.UTC(anio, 0, dia, hh, mm);
}
// los glifos que no se han reconocido se apuntan en el índice para poder añadirlos
function glifosRaros(g) {
  return [25, 26, 27, 28, 29, 31, 32, 33, 34].map((k) => glifo(g.px, g.ancho, 27 + 9 * k)).filter((s) => !Object.values(RAD_CIFRAS).includes(s));
}

// colores de Rain Alarm de cada punto del círculo del radar (480×480; 0 = sin lluvia)
function nivelesRadar(g) {
  const N = 480, col = new Int16Array(256).fill(-1), amarillo = new Uint8Array(256), dbz = new Int16Array(256);
  for (let i = 0; i < g.pal.length / 3; i++) {
    const k = g.pal[3 * i] + "," + g.pal[3 * i + 1] + "," + g.pal[3 * i + 2];
    if (RAD_DBZ[k] !== undefined) { col[i] = mmhCol(dbzMmh(RAD_DBZ[k])); dbz[i] = RAD_DBZ[k]; }
    if (k === "255,255,0") amarillo[i] = 1;
  }
  const niv = new Uint8Array(N * N), dudoso = new Uint8Array(N * N);
  for (let y = 0; y < N; y++) for (let x = 0; x < N; x++) {
    if ((x - 240) ** 2 + (y - 240) ** 2 > 239 * 239) continue;
    const i = g.px[y * g.ancho + x];
    if (amarillo[i]) {
      // el amarillo (48 dBZ) es también el color de las fronteras: solo es lluvia si al lado hay lluvia fuerte
      let fuerte = false;
      for (let dy = -1; dy <= 1 && !fuerte; dy++) for (let dx = -1; dx <= 1; dx++) {
        const j = g.px[(y + dy) * g.ancho + x + dx];
        if (!amarillo[j] && col[j] > 0 && dbz[j] >= 36) { fuerte = true; break; }
      }
      if (fuerte) niv[y * N + x] = col[i];
      else dudoso[y * N + x] = 1;
    } else if (col[i] > 0) niv[y * N + x] = col[i];
  }
  // debajo de las rayas de las fronteras se pone la lluvia de alrededor
  for (let y = 1; y < N - 1; y++) for (let x = 1; x < N - 1; x++) {
    if (!dudoso[y * N + x]) continue;
    let m = 0;
    for (let dy = -1; dy <= 1; dy++) for (let dx = -1; dx <= 1; dx++) { const j = (y + dy) * N + x + dx; if (!dudoso[j] && niv[j] > m) m = niv[j]; }
    niv[y * N + x] = m;
  }
  return niv;
}

// qué punto de cada radar cae en cada punto de la imagen del mapa (se calcula una vez)
let radMalla = null;
function mallaRadar() {
  if (radMalla) return radMalla;
  const c = RAD_CAJA, w = RAD_ANCHO, R = 6371, rad = Math.PI / 180;
  const merc = (lat) => Math.log(Math.tan(Math.PI / 4 + (lat * rad) / 2));
  const yN = merc(c.n), yS = merc(c.s);
  const h = Math.round((w * (yN - yS)) / ((c.e - c.o) * rad));
  const tablas = RAD_RADARES.map(() => new Int32Array(w * h).fill(-1));
  for (let r = 0; r < h; r++) {
    const lat = Math.atan(Math.sinh(yN - ((r + 0.5) / h) * (yN - yS)));
    for (let k = 0; k < w; k++) {
      const lon = (c.o + ((k + 0.5) / w) * (c.e - c.o)) * rad;
      RAD_RADARES.forEach((rd, n) => {
        const p0 = rd.lat * rad, dl = lon - rd.lon * rad;
        const cc = Math.min(1, Math.max(-1, Math.sin(p0) * Math.sin(lat) + Math.cos(p0) * Math.cos(lat) * Math.cos(dl)));
        const ang = Math.acos(cc), f = ang > 1e-12 ? ang / Math.sin(ang) : 1;
        let X = R * f * Math.cos(lat) * Math.sin(dl), Y = R * f * (Math.cos(p0) * Math.sin(lat) - Math.sin(p0) * Math.cos(lat) * Math.cos(dl));
        const cg = Math.cos(rd.giro), sg = Math.sin(rd.giro);
        [X, Y] = [cg * X - sg * Y, sg * X + cg * Y];
        const px = Math.round(240 + X / rd.km), py = Math.round(240 - Y / rd.km);
        if ((px - 240) ** 2 + (py - 240) ** 2 <= 239 * 239) tablas[n][r * w + k] = py * 480 + px;
      });
    }
  }
  radMalla = { w, h, tablas };
  return radMalla;
}
// junta los radares (en cada punto, el que más lluvia ve) en una imagen con la proyección del mapa
function componer(niveles) {
  const m = mallaRadar(), idx = new Uint8Array(m.w * m.h);
  let lluvia = 0;
  for (let i = 0; i < idx.length; i++) {
    let v = 0;
    for (let n = 0; n < niveles.length; n++) { const t = m.tablas[n][i]; if (t >= 0 && niveles[n] && niveles[n][t] > v) v = niveles[n][t]; }
    idx[i] = v;
    if (v) lluvia++;
  }
  return { w: m.w, h: m.h, idx, lluvia };
}

// PNG de paleta (0 = transparente, 1…12 = escala de Rain Alarm)
const CRC = (() => { const t = new Uint32Array(256); for (let n = 0; n < 256; n++) { let c = n; for (let k = 0; k < 8; k++) c = c & 1 ? 0xedb88320 ^ (c >>> 1) : c >>> 1; t[n] = c >>> 0; } return t; })();
function trozo(tipo, datos) {
  const out = new Uint8Array(12 + datos.length), dv = new DataView(out.buffer);
  dv.setUint32(0, datos.length);
  for (let i = 0; i < 4; i++) out[4 + i] = tipo.charCodeAt(i);
  out.set(datos, 8);
  let c = 0xffffffff;
  for (let i = 4; i < 8 + datos.length; i++) c = CRC[(c ^ out[i]) & 255] ^ (c >>> 8);
  dv.setUint32(8 + datos.length, (c ^ 0xffffffff) >>> 0);
  return out;
}
async function pngPaleta(w, h, idx) {
  const ih = new Uint8Array(13), dv = new DataView(ih.buffer);
  dv.setUint32(0, w); dv.setUint32(4, h); ih[8] = 8; ih[9] = 3;
  const pal = new Uint8Array(3 * (RAD_ESC.length + 1)), trn = new Uint8Array(RAD_ESC.length + 1);
  RAD_ESC.forEach((e, i) => { pal.set([e[1], e[2], e[3]], 3 * (i + 1)); trn[i + 1] = 255; });
  const crudo = new Uint8Array((w + 1) * h);
  for (let y = 0; y < h; y++) crudo.set(idx.subarray(y * w, (y + 1) * w), y * (w + 1) + 1);
  const partes = [new Uint8Array([137, 80, 78, 71, 13, 10, 26, 10]), trozo("IHDR", ih), trozo("PLTE", pal), trozo("tRNS", trn), trozo("IDAT", await desinflar(crudo)), trozo("IEND", new Uint8Array(0))];
  const out = new Uint8Array(partes.reduce((a, p) => a + p.length, 0));
  let o = 0;
  for (const p of partes) { out.set(p, o); o += p.length; }
  return out;
}

const nombreHora = (ms) => new Date(ms).toISOString().replace(/[-:T]/g, "").slice(0, 12);

async function leerIndice(sb) {
  try {
    const { data } = await sb.storage.from(RAD_BUCKET).download("indice.json");
    if (!data) return null;
    return JSON.parse(await data.text());
  } catch (_e) { return null; }
}

// pide a la AEMET la última imagen de un radar regional → { g (GIF leído), hora } o { error }
async function pedirRadar(key, id) {
  let j = null, fallo = "";
  try {
    const r1 = await fetch("https://opendata.aemet.es/opendata/api/red/radar/regional/" + id, { headers: { api_key: key, Accept: "application/json" } });
    const txt = await r1.text();
    if (r1.status === 401 || r1.status === 403) return { error: "clave" };
    if (r1.status === 429) return { error: "limite" };
    try { j = JSON.parse(txt); } catch (_e) { j = null; }
    if (j && j.estado === 429) return { error: "limite" };
    if (j && j.datos && !okAemet(j.datos)) return { error: "aemet", fallo: id + ": enlace rechazado (no es de aemet.es)" };
    if (!j || !j.datos) return { error: "aemet", fallo: id + " http " + r1.status + " · " + txt.slice(0, 200) };
  } catch (e) { return { error: "aemet", fallo: id + " sin respuesta: " + String((e && e.message) || e) }; }
  try {
    const r2 = await fetch(j.datos);
    const b = new Uint8Array(await r2.arrayBuffer());
    if (!r2.ok || b[0] !== 0x47) {
      fallo = "imagen " + id + " http " + r2.status + " · " + new TextDecoder().decode(b.subarray(0, 200)).replace(/\s+/g, " ");
      return { error: /429|l[ií]mite/i.test(fallo) ? "limite" : "aemet", fallo };
    }
    const g = leerGif(b);
    return { g, hora: leerHora(g) };
  } catch (e) { return { error: "formato", fallo: id + ": " + String((e && e.message) || e) }; }
}

// pide a la AEMET las imágenes de los radares y guarda la nueva. Devuelve el índice (o {error}).
async function radarGuardar(sb) {
  const key = Deno.env.get("AEMET_API_KEY");
  if (!key) return { error: "sinclave" };
  radarIntento = Date.now();
  const ind = (await leerIndice(sb)) || { frames: [] };
  if (!Array.isArray(ind.frames)) ind.frames = [];
  const anotar = async (error) => {
    ind.intento = Date.now(); ind.ultimoError = error || "";
    await sb.storage.from(RAD_BUCKET).upload("indice.json", new TextEncoder().encode(JSON.stringify(ind)), { contentType: "application/json", upsert: true, cacheControl: "60" });
  };
  const res = [];
  for (const rd of RAD_RADARES) {
    const r = await pedirRadar(key, rd.id);
    if (r.error) console.error("AEMET radar " + rd.id + ": " + r.error + " " + (r.fallo || ""));
    res.push(r);
    if (r.error === "clave") { await anotar("clave"); return { error: "clave" }; }
  }
  const buenos = res.filter((r) => r.g);
  if (!buenos.length) { const e = res.some((r) => r.error === "limite") ? "limite" : res[0].error || "aemet"; await anotar(e); return { error: e }; }
  // hora: la del pie de la imagen; si no se puede leer, la de ahora menos el retraso habitual de la AEMET (15 min)
  const ahora = Date.now(), estimada = Math.floor((ahora - 15 * 60000) / 600000) * 600000;
  const raros = [];
  buenos.forEach((r) => {
    if (!r.hora || r.hora > ahora + 5 * 60000 || r.hora < ahora - 90 * 60000) { raros.push(...glifosRaros(r.g)); r.hora = estimada; }
  });
  ind.glifos = raros.slice(0, 9);
  const hora = Math.max(...buenos.map((r) => r.hora));
  const nombre = nombreHora(hora) + ".png";
  ind.caja = [[RAD_CAJA.s, RAD_CAJA.o], [RAD_CAJA.n, RAD_CAJA.e]];
  let nuevas = 0;
  if (!ind.frames.some((f) => f.f === nombre)) {
    // cada radar solo entra si su imagen es de esa hora (o 10 min antes, si va con retraso)
    const niveles = res.map((r) => (r.g && hora - r.hora <= 10 * 60000 ? nivelesRadar(r.g) : null));
    const rc = componer(niveles);
    const png = await pngPaleta(rc.w, rc.h, rc.idx);
    const up = await sb.storage.from(RAD_BUCKET).upload(nombre, png, { contentType: "image/png", upsert: true, cacheControl: "31536000" });
    if (up && up.error) { console.error("radar guardar: " + up.error.message); await anotar("guardar"); return { error: "guardar" }; }
    ind.frames.push({ t: hora, f: nombre, lluvia: rc.lluvia, radares: res.map((r, n) => (niveles[n] ? RAD_RADARES[n].id : "")).filter(Boolean).join(",") });
    nuevas = 1;
  }
  ind.frames.sort((a, b) => a.t - b.t);
  const limite = ind.frames[ind.frames.length - 1].t - RAD_MIN * 60000;
  const viejos = ind.frames.filter((f) => f.t < limite).map((f) => f.f);
  ind.frames = ind.frames.filter((f) => f.t >= limite);
  if (viejos.length) await sb.storage.from(RAD_BUCKET).remove(viejos);
  ind.hecho = Date.now();
  ind.nuevas = nuevas;
  await anotar(res.some((r) => r.error) ? "parcial" : "");
  return ind;
}

// la app pide el radar: normalmente lo trae la tarea programada; si lleva más de 25 min sin intentarlo, lo pide la app
async function radar(sb) {
  let ind = await leerIndice(sb);
  const ult = Math.max((ind && ind.intento) || 0, (ind && ind.hecho) || 0);
  if (Date.now() - ult > 25 * 60000 && Date.now() - radarIntento > 10 * 60000) {
    const r = await radarGuardar(sb);
    if (!r.error) ind = r;
  }
  if (!ind || !ind.frames || !ind.frames.length) return out({ error: (ind && ind.ultimoError) || "aemet" });
  return out({ ok: true, caja: ind.caja, frames: ind.frames.map((f) => ({ t: f.t, f: f.f })) });
}

async function contar(sb) {
  const { data } = await sb.from("datos").select("user_id,actualizado,contenido");
  const m = {};
  (data || []).forEach((d) => {
    const c = d.contenido || {};
    m[d.user_id] = {
      fincas: Array.isArray(c.fincas) ? c.fincas.length : 0,
      recintos: c.recintos ? Object.keys(c.recintos).length : 0,
      actualizado: d.actualizado || null,
    };
  });
  return m;
}

Deno.serve(async (req) => {
  const origen = req.headers.get("Origin") || "";
  CORS = { ...CORS, "Access-Control-Allow-Origin": WEB };
  if (req.method === "OPTIONS") return new Response("ok", { headers: CORS });
  if (origen && origen !== WEB) return out({ error: "origen" }, 403);
  try {
    const sb = createClient(Deno.env.get("SUPABASE_URL"), Deno.env.get("SUPABASE_SERVICE_ROLE_KEY"), {
      auth: { persistSession: false, autoRefreshToken: false },
    });
    // tarea programada de Supabase (cada 10 min): guarda la imagen nueva del radar. Su clave está en la tabla radar_cron.
    const clCron = req.headers.get("x-radar-cron") || "";
    if (clCron) {
      const { data: rc } = await sb.from("radar_cron").select("clave").eq("id", 1).maybeSingle();
      if (!rc || !rc.clave || rc.clave !== clCron) return out({ error: "prohibido" }, 403);
      const r = await radarGuardar(sb);
      return out(r.error ? { error: r.error } : { ok: true, imagenes: r.frames.length });
    }
    const jwt = (req.headers.get("Authorization") || "").replace(/^Bearer\s+/i, "");
    const { data: ud } = await sb.auth.getUser(jwt);
    const user = ud && ud.user;
    if (!user) return out({ error: "nologin" }, 401);
    let b = {};
    try { b = await req.json(); } catch (_e) { /* vacío */ }
    const a = String(b.accion || "");

    if (a === "radar") return await radar(sb);

    // a partir de aquí, solo Ramon
    if (String(user.email || "").toLowerCase() !== ADMIN) return out({ error: "prohibido" }, 403);

    if (a === "radar-info") {
      // una sola petición a un radar regional (mu, va…) y un informe de lo que llega
      const key = Deno.env.get("AEMET_API_KEY");
      if (!key) return out({ error: "sinclave" });
      const id = /^[a-z]{2}$/.test(String(b.radar || "")) ? String(b.radar) : "mu";
      const r = await pedirRadar(key, id);
      const lineas = ["radar " + id + ": " + (r.error ? "fallo " + r.error + " " + (r.fallo || "") : "imagen " + r.g.ancho + "×" + r.g.alto)];
      if (r.g) {
        lineas.push("hora leída: " + (r.hora ? new Date(r.hora).toISOString() : "no se pudo leer") + " · cifras sin reconocer: " + glifosRaros(r.g).length);
        const niv = nivelesRadar(r.g);
        let n = 0;
        niv.forEach((v) => { if (v) n++; });
        lineas.push("puntos con lluvia: " + n);
      }
      return out({ ok: true, mime: "prueba", bytes: 0, lm: "", meta: lineas.join("\n") });
    }

    if (a === "listar") {
      const { data, error } = await sb.auth.admin.listUsers({ page: 1, perPage: 200 });
      if (error) return out({ error: "fallo" });
      const cuentas = await contar(sb);
      const lista = data.users.map((u) => {
        const c = cuentas[u.id] || {};
        const bloq = u.banned_until && new Date(u.banned_until).getTime() > Date.now();
        return {
          id: u.id,
          nombre: String(u.email || "").replace(DOMINIO, ""),
          admin: String(u.email || "").toLowerCase() === ADMIN,
          creado: u.created_at,
          ultimo: u.last_sign_in_at || null,
          bloqueado: !!bloq,
          fincas: c.fincas || 0,
          recintos: c.recintos || 0,
          actualizado: c.actualizado || null,
        };
      });
      return out({ ok: true, usuarios: lista });
    }

    if (a === "crear") {
      const nombre = String(b.nombre || "").trim().toLowerCase();
      const pass = String(b.pass || "");
      if (!/^[a-z0-9][a-z0-9._-]{1,29}$/.test(nombre)) return out({ error: "nombre" });
      if (pass.length < PASS_MIN) return out({ error: "pass" });
      const { data, error } = await sb.auth.admin.createUser({ email: nombre + DOMINIO, password: pass, email_confirm: true });
      if (error) {
        return out({ error: /already|registered|exists/i.test(error.message || "") ? "existe" : "fallo" });
      }
      return out({ ok: true, id: data.user.id });
    }

    const id = String(b.id || "");
    if (!/^[0-9a-f-]{36}$/.test(id)) return out({ error: "id" });
    const { data: tgt } = await sb.auth.admin.getUserById(id);
    if (!tgt || !tgt.user) return out({ error: "noexiste" });
    const esAdmin = String(tgt.user.email || "").toLowerCase() === ADMIN;

    if (a === "clave") {
      const pass = String(b.pass || "");
      if (pass.length < PASS_MIN) return out({ error: "pass" });
      const { error } = await sb.auth.admin.updateUserById(id, { password: pass });
      return error ? out({ error: "fallo" }) : out({ ok: true });
    }
    if (a === "bloquear") {
      if (esAdmin) return out({ error: "propio" });
      const { error } = await sb.auth.admin.updateUserById(id, { ban_duration: b.bloquear ? "876000h" : "none" });
      return error ? out({ error: "fallo" }) : out({ ok: true });
    }
    if (a === "borrar") {
      if (esAdmin) return out({ error: "propio" });
      for (const t of ["enlace_comentarios", "enlaces", "fotos", "datos_hist", "datos"]) {
        await sb.from(t).delete().eq("user_id", id);
      }
      const { error } = await sb.auth.admin.deleteUser(id);
      return error ? out({ error: "fallo" }) : out({ ok: true });
    }
    if (a === "ver") {
      const { data } = await sb.from("datos").select("contenido").eq("user_id", id).maybeSingle();
      return out({ ok: true, nombre: String(tgt.user.email || "").replace(DOMINIO, ""), doc: (data && data.contenido) || null });
    }
    if (a === "foto") {
      const fid = String(b.foto || "");
      const { data } = await sb.from("fotos").select("contenido").eq("id", fid).eq("user_id", id).maybeSingle();
      return out({ ok: true, contenido: (data && data.contenido) || null });
    }
    if (a === "doc") {
      // Entrega 95: un documento de otro usuario, para su copia completa (solo Ramon)
      const did = String(b.doc || "");
      if (!/^[A-Za-z0-9._-]{1,100}$/.test(did)) return out({ error: "doc" });
      const { data, error } = await sb.storage.from("documentos").download(id + "/" + did);
      if (error || !data) return out({ ok: true, b64: null });
      const u8 = new Uint8Array(await data.arrayBuffer());
      let s = "";
      for (let i = 0; i < u8.length; i += 0x8000) s += String.fromCharCode.apply(null, Array.from(u8.subarray(i, i + 0x8000)));
      return out({ ok: true, mime: data.type || "", b64: btoa(s) });
    }
    return out({ error: "accion" }, 400);
  } catch (e) {
    console.error("servidor: " + String((e && e.message) || e));
    return out({ error: "fallo" }, 500);
  }
});
